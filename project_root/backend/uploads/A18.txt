pi = 3.14
 	
def square(num):
    """
    Function to square a number.
    
    Args:
        num (int or float): The number to be squared.
    
    Returns:
        int or float: The square of the number.
    """
    return num ** 2

def product(a, b, c):
    """
    Function to calculate the product of three numbers.
    
    Args:
        a (int or float): The first number.
        b (int or float): The second number.
        c (int or float): The third number.
    
    Returns:
        int or float: The product of the three numbers.
    """
    return a * b * c
