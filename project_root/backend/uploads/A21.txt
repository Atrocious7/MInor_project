import os
import webbrowser


print("Choices:")
print("1.Open Notepad.")
print("2.Open Calculator")
print("3.Search given item on web browser")
print("4.Exit")

choice = int(input("Enter your choice: "))

if choice == 1:
    os.system("Notepad")
elif choice == 2:
    os.system("calc")
elif choice == 3:
    item = input("Enter the item you want to search: ")
    webbrowser.open("https://www.google.com/search?q=" + item)
elif choice == 4:
    print("Exiting now...")
    exit()
else:
    print("Invalid choice")

 def meditation_time():

 def exercise_time():

 total_minutes = hours*60

 minutes = total_minutes
 while minutes > 0:
     if minutes % 20 == 0:
         if minutes <= 20:
           relax_time()
        elif 20 < minutes <= 40:
            meditation_time()
         else:
