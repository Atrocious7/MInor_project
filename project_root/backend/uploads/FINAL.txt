'''
Create a class Doctor
1. Make a constructor that take doctor name and qualification as
argument and fee as input.
2. Make a function msg, print expensive doctor if fee is more than
1000.
3. Make function display that print all details of doctor
4. Make destructor that display have a nice day

'''

class Doctor:
    def __init__(self, name, qualification, fee):
        self.name = name
        self.qualification = qualification
        self.fee = fee
    
    def msg(self):
        if self.fee > 1000:
            print("expensive doctor")
    
    def display(self):
        print("Name:", self.name)
        print("Qualification:", self.qualification)
        print("Fee:", self.fee)
    
    def __del__(self):
        print("have a nice day")


name = input("Enter the doctor's name: ")
qualification = input("Enter the doctor's qualification: ")
fee = float(input("Enter the doctor's fee: "))

d = Doctor(name, qualification, fee)
d.msg()
d.display()
del d
