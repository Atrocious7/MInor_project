names = []
while True:
    name = input("Enter name (or 'done' to finish): ")
    if name.lower() == 'done':
        break
    names.append(name)

# Write names to file
with open('friend book', 'w') as f:
    for name in names:
        f.write(name + '\n')

# Search name in file
search_name = input("Enter a name to search: ")
found = False
with open('friend book', 'r') as f:
    for line in f:
        if line.strip() == search_name:
            print(f"'{search_name}' is present in the file")
            found = True
            break
if not found:
    print(f"'{search_name}' is not present in the file")

# Display names that starts with 'j'
with open('friend book', 'r') as f:
    for line in f:
        if line.strip().startswith('j'):
            print(line.strip())
