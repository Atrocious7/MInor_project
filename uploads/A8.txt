number = int(input("Enter the number for which you want to display the multiplication table: "))
num_range = int(input("Enter the range up to which you want to display the multiplication table: "))


print("\nMultiplication Table for", number, "up to", num_range, ":")

for i in range(1, num_range + 1):
    result = number * i
    print(number, "x", i, "=", result)
